
#include <iostream>
#include <string>
using namespace std;

class SavingAccount {
    string name;
    int accountNumber;
    double balance, rate;

public:
    SavingAccount(string n, int a, double b, double r)
        : name(n), accountNumber(a), balance(b), rate(r) {}

    void deposit(double amount) {
        if (amount > 0)
            balance += amount;
    }

    void withdraw(double amount) {
        if (amount > 0 && amount <= balance)
            balance -= amount;
        else
            cout << "Insufficient balance!\n";
    }

    void applyInterest() {
        balance += balance * rate / 100;
    }

    void display() {
        cout << "\nSavings Account\n";
        cout << "Name: " << name << endl;
        cout << "Account Number: " << accountNumber << endl;
        cout << "Balance: " << balance << endl;
        cout << "Interest Rate: " << rate << "%\n";
    }
};

class CheckingAccount {
    string name;
    int accountNumber;
    double balance, fee;

public:
    CheckingAccount(string n, int a, double b, double f)
        : name(n), accountNumber(a), balance(b), fee(f) {}

    void deposit(double amount) {
        if (amount > 0)
            balance += amount;
    }

    void withdraw(double amount) {
        if (amount > 0 && amount + fee <= balance)
            balance -= amount + fee;
        else
            cout << "Insufficient balance!\n";
    }

    void display() {
        cout << "\nChecking Account\n";
        cout << "Name: " << name << endl;
        cout << "Account Number: " << accountNumber << endl;
        cout << "Balance: " << balance << endl;
        cout << "Transaction Fee: " << fee << endl;
    }
};

int main() {
    SavingAccount s("Asha", 1001, 5000, 3);
    CheckingAccount c("Rahul", 1002, 3000, 20);

    s.deposit(1000);
    s.withdraw(500);
    s.applyInterest();
    s.display();

    c.deposit(500);
    c.withdraw(1000);
    c.display();

    return 0;
}
